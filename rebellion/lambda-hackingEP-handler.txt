  let body = {};
  try { body = typeof event.body === 'string' ? JSON.parse(event.body || '{}') : (event.body || {}); }
  catch { return respond(400, { ok: false, message: 'bad json' }); }

}

// ── Path utils ──────────────────────────────────────────────────────────
function normPath(p) {
  const parts = String(p || '').split('/').filter(Boolean);
  const r = [];
  for (const pt of parts) {
    if (pt === '.') continue;
    if (pt === '..') { r.pop(); continue; }
    r.push(pt);
  }
  return '/' + r.join('/');
}
function resolvePath(state, input, server) {
  const HOME = server === 'core01' ? '/home/observer' : '/root';
  if (!input || input === '~') return HOME;
  if (input.startsWith('~/'))  return normPath(HOME + '/' + input.slice(2));
  if (input.startsWith('/'))   return normPath(input);
  return normPath(state.cwd + '/' + input);
}
function getEntry(fs, path) {
  const p = path === '/' ? '/' : path.replace(/\/+$/, '');
  if (fs[p]) return fs[p];
  if (p === '/') return fs[p] || null;
  const idx = p.lastIndexOf('/');
  const parent = idx <= 0 ? '/' : p.slice(0, idx);
  const name = p.slice(idx + 1);
  const parentEntry = fs[parent];
  if (parentEntry && parentEntry.type === 'dir' && (parentEntry.ch || []).includes(name)) return { type: 'file' };
  return null;
}

  return undefined;
}

// ── Tiny shell parser ───────────────────────────────────────────────────
function tokenize(s) {
  const out = []; let i = 0, cur = '', quote = null;
  while (i < s.length) {
    const c = s[i];
    if (quote) { if (c === quote) { quote = null; i++; continue; } cur += c; i++; continue; }
    if (c === '"' || c === "'") { quote = c; i++; continue; }
    if (c === ' ' || c === '\t') { if (cur) { out.push(cur); cur = ''; } i++; continue; }
    cur += c; i++;
  }
  if (cur) out.push(cur);
  return out;
}
function splitPipes(raw) {
  const out = []; let cur = '', quote = null;
  for (let i = 0; i < raw.length; i++) {
    const c = raw[i];
    if (quote) { cur += c; if (c === quote) quote = null; continue; }
    if (c === '"' || c === "'") { quote = c; cur += c; continue; }
    if (c === '|') { out.push(cur.trim()); cur = ''; continue; }
    cur += c;
  }
  if (cur.trim()) out.push(cur.trim());
  return out;
}
function splitCommands(raw) {
  const out = []; let cur = '', quote = null;
  for (let i = 0; i < raw.length; i++) {
    const c = raw[i], n = raw[i + 1];
    if (quote) { cur += c; if (c === quote) quote = null; continue; }
    if (c === '"' || c === "'") { quote = c; cur += c; continue; }
    if (c === ';') { if (cur.trim()) out.push(cur.trim()); cur = ''; continue; }
    if (c === '&' && n === '&') { if (cur.trim()) out.push(cur.trim()); cur = ''; i++; continue; }
    cur += c;
  }
  if (cur.trim()) out.push(cur.trim());
  return out;
}
function hasGlob(s) {
  return /[*?[]/.test(String(s || ''));
}
function globToRegExp(glob) {
  let out = '^';
  for (const ch of glob) {
    if (ch === '*') out += '.*';
    else if (ch === '?') out += '.';
    else out += ch.replace(/[|\\{}()[\]^$+?.]/g, '\\$&');
  }
  return new RegExp(out + '$');
}
function listDirNames(store, dir, showHidden = true) {
  const ent = getEntry(store.FS, dir);
  if (!ent || ent.type !== 'dir') return null;
  let names = [...(ent.ch || [])];
  if (showHidden) names = ['.', '..', ...names];
  return names;
}
function expandPathArgs(store, state, args) {
  const out = [];
  for (const arg of args) {
    if (!hasGlob(arg)) { out.push(arg); continue; }
    const slash = arg.lastIndexOf('/');
    const dirPart = slash >= 0 ? arg.slice(0, slash + 1) : '';
    const baseGlob = slash >= 0 ? arg.slice(slash + 1) : arg;
    const dirPath = resolvePath(state, dirPart || '.', store.SERVER);
    const names = listDirNames(store, dirPath, false) || [];
    const re = globToRegExp(baseGlob);
    const matches = names.filter(n => re.test(n)).sort().map(n => dirPart + n);
    if (matches.length) out.push(...matches);
    else out.push(arg);
  }
  return out;
}
function readArgText(store, state, arg) {
  const t = resolvePath(state, arg, store.SERVER);
  if (t === '/tmp/db_017.sql') {
    if (!state.decrypted) return { error: `No such file or directory` };
    return { text: store.DB017_SQL || '' };
  }
  const content = fileContent(store, t);
  if (content !== undefined) return { text: content };
  const e = getEntry(store.FS, t);
  if (!e) return { error: 'No such file or directory' };
  if (e.type === 'dir') return { error: 'Is a directory' };
  return { error: 'Permission denied' };
}
function commandPath(store, cmd) {
  const dirs = ['/bin', '/usr/bin', '/usr/local/bin', '/sbin', '/usr/sbin'];
  for (const dir of dirs) {
    const ent = getEntry(store.FS, dir);
    if (ent && (ent.ch || []).includes(cmd)) return `${dir}/${cmd}`;
  }
  const implemented = ['cd','pwd','echo','history','help','exit','logout','type','which','ps','netstat','df','ss','lsof','systemctl','sqlite3','decrypt','tree','grep','head','tail','find','wc','sort','uniq','base64','stat','date','whoami','hostname','uname','id','uptime','env','ifconfig','iptables','last','lastb','iostat','nmap','ping'];
  if (implemented.includes(cmd) || (store.CMDS || {})[cmd] !== undefined) return `/usr/bin/${cmd}`;
  return '';
}
function walkFiles(store, start, includeDirs = false) {
  const root = resolvePath({ cwd: '/', prevCwd: '/' }, start, store.SERVER);
  const out = [];
  function rec(path) {
    const ent = getEntry(store.FS, path);
    if (ent && ent.type === 'dir') {
      if (includeDirs) out.push(path);
      for (const name of (ent.ch || [])) rec(path === '/' ? '/' + name : path + '/' + name);
      return;
    }
    if (ent || fileContent(store, path) !== undefined) out.push(path);
  }
  rec(root);
  return out;
}
function sshHostFromArgs(args) {
  const skipValue = new Set(['-F', '-i', '-p', '-l', '-o', '-J', '-W', '-b']);
  let host = '';
  for (let i = 0; i < args.length; i++) {
    const arg = args[i];
    if (!arg) continue;
    if (skipValue.has(arg)) { i++; continue; }
    if (arg.startsWith('-')) continue;
    host = arg;
  }
  if (!host) return '';
  if (host.includes('@')) host = host.split('@').pop();
  return host.replace(/^\[|\]$/g, '').replace(/:\d+$/, '');
}
function sshRouteFor(store, args) {
  const host = sshHostFromArgs(args);
  const tail = args.join(' ');
  if (!store.SSH_ROUTES) return null;
  for (const [routeHost, route] of Object.entries(store.SSH_ROUTES)) {
    const aliases = [routeHost, ...(route.aliases || [])];
    if (aliases.some(alias => host === alias || tail.includes(alias))) return { host: routeHost, route };
  }
  return null;
}

function runShell({ store, raw, state }) {
  // SSH y/N wait flow
  if (state.sshWait) {
    const ans = raw.trim().toLowerCase();
    if (ans === 'y' || ans === 'yes') {
      const target = state.sshTarget || 'server2.html';
      state.sshWait = false; state.sshTarget = null;
      return { output: 'Connecting...', action: 'ssh', target, state };
    }
    state.sshWait = false; state.sshTarget = null;
    return { output: 'Aborted.', state };
  }

  if (state.sqliteMode) return runSqlite({ store, raw, state });

  const commands = splitCommands(raw);
  if (commands.length > 1) {
    const outputs = [];
    const reactions = [];
    for (const command of commands) {
      const result = runShell({ store, raw: command, state });
      state = result.state || state;
      if (result.output) outputs.push(result.output);
      if (Array.isArray(result.reactions)) reactions.push(...result.reactions);
      if (result.action) return Object.assign({ output: outputs.join('\n'), state, reactions }, result);
    }
    return { output: outputs.join('\n'), state, reactions };
  }

  const stages = splitPipes(raw);
  if (stages.length === 0) return { output: '', state };

  let stdin = null, lastOut = '';
  const reactions = [];
  for (let s = 0; s < stages.length; s++) {
    const stage = stages[s];
    const result = runOne({ store, raw: stage, state, stdin });
    if (Array.isArray(result.reactions)) reactions.push(...result.reactions);
    if (s === stages.length - 1 && result.action) {
      return Object.assign({ state, reactions }, result);
    }
    stdin = result.output;
    lastOut = result.output;
  }
  return { output: lastOut, state, reactions };
}

function runOne({ store, raw, state, stdin }) {
  const trimmed = raw.trim();
  if (!trimmed) return { output: '' };
  const tokens = tokenize(trimmed);
  const cmd = tokens[0];
  const rest = tokens.slice(1);

  if (trimmed.toLowerCase() === STORY_CONFIG.RESISTANCE_COMMAND)
    return { output: resistanceReveal() };

  if (cmd === 'clear' || cmd === 'reset') return { action: 'clear', output: '' };

  // SSH
  if (cmd === 'ssh') {
    const match = sshRouteFor(store, rest);
    if (match) {
      state.sshWait = true;
      state.sshTarget = match.route.target;
      return { output: match.route.banner };
    }
    return { output: `ssh: connect to host ${sshHostFromArgs(rest) || rest[0] || '?'}: Connection refused` };
  }

  if (cmd === 'cd') {
    if (rest.length > 1) return { output: 'bash: cd: too many arguments' };
    const target = rest[0] === '-' ? state.prevCwd : resolvePath(state, rest[0], store.SERVER);
    const entry  = getEntry(store.FS, target);
    if (!entry)                return { output: `bash: cd: ${rest[0] || '~'}: No such file or directory` };
    if (entry.type === 'file') return { output: `bash: cd: ${rest[0]}: Not a directory` };
    const old = state.cwd; state.cwd = target; state.prevCwd = old;
    return { output: rest[0] === '-' ? state.cwd : '' };
  }
  if (cmd === 'pwd') return { output: state.cwd };

  if (cmd === 'ls') {
    // Forensic tell: the beacon's exe was unlinked while still running, so
    // `ls -l /proc/<pid>/exe` shows the symlink pointing at "(deleted)".
    const procExe = rest.find(a => /^\/proc\/\d+\/exe$/.test(a));
    if (procExe && store.SERVER === 'node083') {
      const pid = procExe.match(/\/proc\/(\d+)\/exe/)[1];
      if (pid === String(state.beaconPid) || pid === '4291') {
        return /-[a-z]*l/.test(rest.join(''))
          ? { output: `lrwxrwxrwx 1 root root 0 May 17 03:30 ${procExe} -> '/usr/bin/python3.10 (deleted)'` }
          : { output: procExe };
      }
      return { output: `ls: cannot access '${procExe}': No such file or directory` };
    }
    const flags = rest.filter(a => a.startsWith('-')).join('');
    const paths = expandPathArgs(store, state, rest.filter(a => !a.startsWith('-')));
    const targets = paths.length ? paths : ['.'];
    const blocks = [];
    let hasDirBlock = false;
    for (const p of targets) {
      const target = resolvePath(state, p, store.SERVER);
      const res = lsOutput(store, target, flags);
      if (res === null) blocks.push(`ls: cannot access '${p}': No such file or directory`);
      else {
        const entry = getEntry(store.FS, target);
        const label = targets.length > 1 && entry && entry.type === 'dir';
        if (label) hasDirBlock = true;
        blocks.push(label ? `${p}:\n${res}` : res);
      }
    }
    return { output: blocks.join(hasDirBlock ? '\n\n' : (/-[a-z]*l/.test(flags) ? '\n' : '  ')) };
  }

  if (cmd === 'tree') {
    const showHidden = rest.includes('-a');
    let levelLimit = Infinity;
    const lIdx = rest.indexOf('-L');
    if (lIdx >= 0 && rest[lIdx + 1] && /^\d+$/.test(rest[lIdx + 1])) levelLimit = parseInt(rest[lIdx + 1], 10);
    const startArg = rest.find(a => !a.startsWith('-') && !/^\d+$/.test(a));
    const start = startArg ? resolvePath(state, startArg, store.SERVER) : state.cwd;
    const e = getEntry(store.FS, start);
    if (!e) return { output: `tree: ${startArg || start} [error opening dir]` };
    const lines = []; let dirs = 0, files = 0;
    lines.push(start);
    function walk(path, depth, prefix) {
      if (depth > levelLimit) return;
      const ent = getEntry(store.FS, path);
      if (!ent || ent.type !== 'dir') return;
      let ch = [...(ent.ch || [])]; if (!showHidden) ch = ch.filter(x => !x.startsWith('.'));
      ch.sort();
      ch.forEach((name, i) => {
        const last = i === ch.length - 1;
        const childAbs = path === '/' ? '/' + name : path + '/' + name;
        const child = getEntry(store.FS, childAbs);
        const branch = last ? '`-- ' : '|-- ';
        const isDir = child && child.type === 'dir';
        if (isDir) dirs++; else files++;
        lines.push(prefix + branch + name + (isDir ? '/' : ''));
        if (isDir) walk(childAbs, depth + 1, prefix + (last ? '    ' : '|   '));
      });
    }
    walk(start, 1, '');
    lines.push('', `${dirs} directories, ${files} files`);
    return { output: lines.join('\n') };
  }

  if (cmd === 'cat') {
    if (stdin !== null && rest.length === 0) return { output: stdin };
    if (rest.length === 0) return { output: 'cat: missing file operand' };
    const out = [];
    for (const arg of expandPathArgs(store, state, rest)) {
      const t = resolvePath(state, arg, store.SERVER);
      if (t === '/tmp/db_017.sql') {
        if (!state.decrypted) { out.push(`cat: ${arg}: No such file or directory`); continue; }
        out.push(store.DB017_SQL || '(empty)'); continue;
      }
      const content = fileContent(store, t);
      if (content !== undefined) { out.push(content); continue; }
      const e = getEntry(store.FS, t);
      if (!e)                    out.push(`cat: ${arg}: No such file or directory`);
      else if (e.type === 'dir')  out.push(`cat: ${arg}: Is a directory`);
      else                        out.push(`cat: ${arg}: Permission denied`);
    }
    // Reading the breach log is the moment the other analysts realise
    // someone is live on the box with them.
    const readsBreach = expandPathArgs(store, state, rest)
      .some(a => resolvePath(state, a, store.SERVER) === '/var/log/breach.log');
    if (store.SERVER === 'node083' && readsBreach && !state.ledgerSeen) {
      state.ledgerSeen = true;
      return { output: out.join('\n'), state, reactions: [
        { delay: 7000, c: 'red', t: `<r.takeda> the breach log is being read right now, this second. who is logged into 083.` },
      ] };
    }
    return { output: out.join('\n') };
  }

  if (cmd === 'head' || cmd === 'tail') {
    let n = 10;
    const nIdx = rest.indexOf('-n');
    if (nIdx >= 0 && /^\d+$/.test(rest[nIdx + 1] || '')) n = parseInt(rest[nIdx + 1], 10);
    const compactN = rest.find(a => /^-\d+$/.test(a));
    if (compactN) n = Math.abs(parseInt(compactN, 10));
    let text = stdin;
    if (text == null) {
      const f = rest.find((a, i) => !a.startsWith('-') && !(rest[i - 1] === '-n') && !/^\d+$/.test(a));
      if (f) {
        const t = resolvePath(state, f, store.SERVER);
        const content = fileContent(store, t);
        if (content !== undefined) text = content;
        else return { output: `${cmd}: cannot open '${f}' for reading: No such file or directory` };
      } else return { output: '' };
    }
    const lines = String(text).split('\n');
    return { output: (cmd === 'head' ? lines.slice(0, n) : lines.slice(-n)).join('\n') };
  }

  if (cmd === 'grep') {
    const flags = rest.filter(a => a.startsWith('-'));
    const positional = rest.filter(a => !a.startsWith('-'));
    const pat = positional[0]; if (!pat) return { output: 'usage: grep PATTERN [file]' };
    const fileArg = positional[1];
    let text = stdin;
    if (text == null && fileArg) {
      const files = expandPathArgs(store, state, [fileArg]);
      const chunks = [];
      for (const f of files) {
        const r = readArgText(store, state, f);
        if (r.error) return { output: `grep: ${f}: ${r.error}` };
        chunks.push(String(r.text));
      }
      text = chunks.join('\n');
    }
    if (text == null) return { output: '' };
    const ci = flags.some(f => f.includes('i'));
    const inv = flags.some(f => f.includes('v'));
    const ext = flags.some(f => f.includes('E'));
    const re = new RegExp(ext ? pat : pat.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), ci ? 'i' : '');
    const out = String(text).split('\n').filter(l => inv ? !re.test(l) : re.test(l));
    return { output: out.join('\n') };
  }

  if (cmd === 'wc') {
    const files = expandPathArgs(store, state, rest.filter(a => !a.startsWith('-')));
    const flagL = rest.includes('-l'), flagW = rest.includes('-w'), flagC = rest.includes('-c');
    const format = (text, name = '') => {
      const lines = text ? text.split('\n').length : 0;
      const words = text.split(/\s+/).filter(Boolean).length;
      const bytes = text.length;
      const parts = [];
      if (!flagL && !flagW && !flagC) parts.push(lines, words, bytes);
      else { if (flagL) parts.push(lines); if (flagW) parts.push(words); if (flagC) parts.push(bytes); }
      return parts.map(n => String(n).padStart(7, ' ')).join('') + (name ? ' ' + name : '');
    };
    if (stdin != null && files.length === 0) return { output: format(String(stdin)).trimStart() };
    if (files.length === 0) return { output: 'wc: missing file operand' };
    const out = [];
    for (const f of files) {
      const r = readArgText(store, state, f);
      out.push(r.error ? `wc: ${f}: ${r.error}` : format(String(r.text), f));
    }
    return { output: out.join('\n') };
  }
  if (cmd === 'sort' || cmd === 'uniq') {
    const files = expandPathArgs(store, state, rest.filter(a => !a.startsWith('-')));
    let text = String(stdin ?? '');
    if (stdin == null && files.length) {
      const chunks = [];
      for (const f of files) {
        const r = readArgText(store, state, f);
        if (r.error) return { output: `${cmd}: cannot read: ${f}: ${r.error}` };
        chunks.push(String(r.text));
      }
      text = chunks.join('\n');
    }
    const lines = text.split('\n');
    if (cmd === 'sort') return { output: lines.sort().join('\n') };
    return { output: lines.filter((l, i, a) => i === 0 || l !== a[i - 1]).join('\n') };
  }
  if (cmd === 'echo') return { output: rest.join(' ') };
  if (cmd === 'type') return { output: rest.length ? rest.map(c => commandPath(store, c) ? `${c} is ${commandPath(store, c)}` : `bash: type: ${c}: not found`).join('\n') : 'type: usage: type [-afptP] name [name ...]' };
  if (cmd === 'which') return { output: rest.map(c => commandPath(store, c)).filter(Boolean).join('\n') };

  if (cmd === 'stat') {
    if (!rest[0]) return { output: 'stat: missing operand' };
    const t = resolvePath(state, rest[0], store.SERVER);
    const e = getEntry(store.FS, t);
    const content = fileContent(store, t);
    if (!e && content === undefined) return { output: `stat: cannot stat '${rest[0]}': No such file or directory` };
    const meta = (store.FILE_META || {})[t] || {
      size: (content || '').length || 4096,
      mode: e && e.type === 'dir' ? '0755/drwxr-xr-x' : '0644/-rw-r--r--',
      uid: '0/root', gid: '0/root',
      atime: '2025-05-17 04:22:18.000000000 +0000',
      mtime: '2025-05-17 03:00:00.000000000 +0000',
      ctime: '2025-05-17 03:00:00.000000000 +0000',
      birth: '2025-05-17 03:00:00.000000000 +0000',
      inode: 200000 + Math.floor(Math.random() * 700000),
    };
    return { output:
`  File: ${t}
  Size: ${meta.size}\tBlocks: ${Math.ceil(meta.size/512)}\tIO Block: 4096   ${e && e.type === 'dir' ? 'directory' : 'regular file'}
Device: 801h/2049d\tInode: ${meta.inode}\tLinks: 1
Access: (${meta.mode})  Uid: (${meta.uid})  Gid: (${meta.gid})
Access: ${meta.atime}
Modify: ${meta.mtime}
Change: ${meta.ctime}
 Birth: ${meta.birth}`};
  }

  if (cmd === 'base64') {
    const decode = rest.includes('-d') || rest.includes('--decode');
    let input = stdin != null ? String(stdin) : '';
    if (!input) {
      const f = rest.find(a => !a.startsWith('-'));
      if (f) {
        const r = readArgText(store, state, f);
        if (r.error) return { output: `base64: ${f}: ${r.error}` };
        input = String(r.text);
      }
    }
    if (!input) return { output: '' };
    try {
      if (decode) return { output: Buffer.from(input.replace(/\s+/g, ''), 'base64').toString('utf8') };
      return { output: Buffer.from(input, 'utf8').toString('base64') };
    } catch { return { output: 'base64: invalid input' }; }
  }

  if (cmd === 'kill' || cmd === 'pkill' || cmd === 'killall') {
    const pid = rest.find(a => /^\d+$/.test(a));
    const beaconRef = store.SERVER === 'node083' &&
      (pid === String(state.beaconPid) || pid === '4291' ||
       /python3|beacon|4291/.test(rest.join(' ')));
    if (beaconRef) {
      // The process really dies — but the cron watchdog brings it straight
      // back under a fresh PID. Killing the symptom, not the cause.
      const oldPid = state.beaconPid || 4291;
      const newPid = 5117 + Math.floor(Math.random() * 90);
      const firstTime = !state.beaconKilled;
      state.beaconKilled = true;
      state.beaconPid = newPid;
      const reactions = firstTime ? [
        { delay: 8000,  c: 'o',   t: `[systemd] livseg-keepalive.timer: watchdog detected beacon down — invoking respawn` },
        { delay: 11500, c: 'r2',  t: `[netwatch] outbound RESTORED — 172.16.47.83:49077 -> 91.213.8.44:4444 — beacon back as PID ${newPid}` },
        { delay: 15000, c: 'red', t: `<r.takeda> it came right back. that's a cron watchdog, not the process. pull the cron job, not the PID.` },
      ] : [
        { delay: 9000,  c: 'r2',  t: `[netwatch] beacon respawned again — PID ${newPid}. /etc/cron.d is bringing it back every time.` },
      ];
      return { output: `[1]+  Killed    /usr/bin/python3 (pid ${oldPid})`, reactions, state };
    }
    // Non-beacon kill by numeric PID. pkill/killall by name fall through to
    // the per-server deny table below.
    if (cmd === 'kill') {
      if (!pid) return { output: 'kill: usage: kill [-s sigspec | -n signum | -sigspec] pid | jobspec ...' };
      if (pid === '1244') return { output: `kill: (${pid}): Operation not permitted` };
      return { output: `kill: (${pid}): No such process` };
    }
  }

  // Forensic: a running-but-unlinked binary still shows via /proc/<pid>/exe.
  // This is how you prove the beacon's executable was deleted from disk.
  if (cmd === 'readlink') {
    const tgt = rest.find(a => !a.startsWith('-')) || '';
    const m = tgt.match(/^\/proc\/(\d+)\/exe$/);
    if (m && store.SERVER === 'node083' && (m[1] === String(state.beaconPid) || m[1] === '4291')) {
      return { output: `/usr/bin/python3.10 (deleted)` };
    }
    if (m) return { output: `readlink: ${tgt}: No such file or directory` };
    return { output: rest.length ? '' : 'readlink: missing operand' };
  }

  if ((cmd === 'openssl' || cmd === 'decrypt') && raw.includes('db_017')) {
    // Accept both the real openssl incantation and the `decrypt` shorthand.
    // The real form needs the key material; reward players who supply it.
    const usesKey = /key\.pem|-kfile|-k\b|-pass|-in\b/.test(raw);
    if (cmd === 'openssl' && !usesKey) {
      return { output: `openssl: enc: AES-256-CBC requires a key — pass -kfile /tmp/.hidden/key.pem` };
    }
    const firstTime = !state.decrypted;
    state.decrypted = true;
    const reactions = firstTime ? [
      { delay: 6500, c: 'red', t: `<GoodGirl17> 017 just got decrypted. good. we left it for them to read.` },
    ] : [];
    return { action: 'decrypt', output: 'Output written: /tmp/db_017.sql', reactions, state };
  }
  if (cmd === 'decrypt') return { output: 'Usage: decrypt <keyfile> <encrypted_file>\n  e.g. decrypt /tmp/.hidden/key.pem /root/databases/db_017.enc' };
  if (cmd === 'openssl') return { output: 'usage: openssl enc -d -aes-256-cbc -in <file> -out <outfile> -kfile <keyfile>' };

  if (cmd === 'sqlite3') {
    const f = rest[0];
    if (f && f.includes('db_017')) {
      if (!state.decrypted) return { output: `sqlite3: ${f}: unable to open database file — decrypt first` };
      state.sqliteMode = true; state.sqliteFile = f;
      return { output: `SQLite version 3.39.2 2022-07-21 15:24:47\nEnter ".help" for usage hints.` };
    }
    if (!f) return { output: 'Usage: sqlite3 <database_file>' };
    return { output: `sqlite3: ${f}: unable to open database file` };
  }

  if (cmd === 'find') {
    const key = `find ${rest.join(' ')}`;
    if (store.CMDS[key] !== undefined) return { output: store.CMDS[key] };
    const start = rest[0] && !rest[0].startsWith('-') ? rest[0] : '.';
    const startPath = resolvePath(state, start, store.SERVER);
    const nameIdx = rest.indexOf('-name');
    const typeIdx = rest.indexOf('-type');
    const namePat = nameIdx >= 0 ? rest[nameIdx + 1] : null;
    const typeVal = typeIdx >= 0 ? rest[typeIdx + 1] : null;
    const all = walkFiles(store, startPath, typeVal === 'd');
    const re = namePat ? globToRegExp(namePat) : null;
    const out = all.filter(p => {
      const ent = getEntry(store.FS, p);
      if (typeVal === 'f' && ent && ent.type === 'dir') return false;
      if (typeVal === 'd' && (!ent || ent.type !== 'dir')) return false;
      if (re && !re.test(p.split('/').pop())) return false;
      return true;
    });
    return { output: out.join('\n') };
  }

  if (cmd === 'systemctl') {
    const writeAction = rest.find(a => /^(start|stop|restart|enable|disable|kill|reset-failed|daemon-reload|mask|unmask|reload|edit|set-property)$/.test(a));
    if (writeAction) return { output: `Failed to ${writeAction} ${rest[rest.indexOf(writeAction)+1] || ''}: ${store.SERVER === 'node083' ? 'Refused — audit-lock active.' : 'Interactive authentication required.'}` };
    if (rest.includes('--failed')) return { output: store.CMDS['systemctl --failed'] || '0 loaded units listed.' };
    if (rest[0] === 'status') return { output: `● ${rest[1] || 'unit'}.service — service\n     Loaded: loaded\n     Active: failed`};
    return { output: 'systemctl: unrecognized arguments' };
  }

  if (cmd === 'history') {
    const histPath = store.SERVER === 'node083' ? '/root/.bash_history' : '/home/observer/.bash_history';
    const text = fileContent(store, histPath) || '';
    return { output: text.split('\n').map((l, i) => `${String(i + 1).padStart(5, ' ')}  ${l}`).join('\n') };
  }
  if (cmd === 'exit' || cmd === 'logout') return { output: 'logout' };
  if (cmd === 'help') return { output: BASH_HELP };

  // ── reflex commands a real analyst would try (absorb, don't 404) ──
  if (cmd === 'sudo') {
    if (store.SERVER === 'node083') {
      const sub = rest.filter(a => !a.startsWith('-'));
      return { output: sub.length ? runOne({ store, raw: sub.join(' '), state, stdin }).output : 'usage: sudo command' };
    }
    if (rest[0] === '-l' || rest[0] === '-ll')
      return { output: `Matching Defaults entries for observer on cd-0s-core01:\n    env_reset, mail_badpass\n\nUser observer is not allowed to run sudo on cd-0s-core01.\nThis incident has been reported to the administrator.` };
    return { output: `[sudo] password for observer: \nsudo: a password is required` };
  }

  if (cmd === 'sha256sum') {
    const files = expandPathArgs(store, state, rest.filter(a => !a.startsWith('-')));
    if (!files.length) return { output: `sha256sum: missing operand` };
    const lines = files.map(f => {
      const r = readArgText(store, state, f);
      if (r.error) return `sha256sum: ${f}: ${r.error}`;
      const hash = createHash('sha256').update(String(r.text), 'utf8').digest('hex');
      return `${hash}  ${f}`;
    });
    return { output: lines.join('\n') };
  }

  if (cmd === 'md5sum') {
    const files = expandPathArgs(store, state, rest.filter(a => !a.startsWith('-')));
    if (!files.length) return { output: `md5sum: missing operand` };
    return { output: files.map(f => {
      const r = readArgText(store, state, f);
      if (r.error) return `md5sum: ${f}: ${r.error}`;
      return `${createHash('md5').update(String(r.text), 'utf8').digest('hex')}  ${f}`;
    }).join('\n') };
  }

  if (cmd === 'dmesg') {
    return { output: beaconize(store.SERVER === 'node083' ? DMESG_NODE083 : DMESG_CORE01, state) };
  }

  if (cmd === 'journalctl') {
    if (store.SERVER === 'node083')
      return { output: `-- journal unreadable: /var/log/journal on /dev/sdb1 returned I/O error --\nHint: the text logs survived. try /var/log/breach.log and /var/log/firewall.log` };
    return { output: `-- Logs begin at Sat 2025-05-17 03:00:00 UTC. --\nMay 17 04:17:33 cd-0s-core01 netwatch[28901]: ALERT: unusual outbound traffic on node-083\nMay 17 04:18:44 cd-0s-core01 netwatch[28901]: CRITICAL: IDS on 172.16.47.83 OFFLINE\n-- full daemon journal forwarded to /var/log/syslog --` };
  }

  if (cmd === 'free') {
    return { output: store.SERVER === 'node083'
      ? `               total        used        free      shared  buff/cache   available\nMem:        16383844    16201232      102400        4096       80212      187392\nSwap:        2097148     2097148           0\n[WARNING: memory pressure critical — oom-killer active]`
      : `               total        used        free      shared  buff/cache   available\nMem:        16383844     7301120     4832192      102400     4250532     7200640\nSwap:        2097148           0     2097148` };
  }

  if (cmd === 'w') {
    if (store.CMDS.w) return { output: store.CMDS.w };
    return { output: ` 04:22:51 up 4:22,  1 user,  load average: 9.87, 9.91, 9.88\nUSER     TTY      FROM             LOGIN@   IDLE   JCPU   PCPU WHAT\nroot     pts/0    10.0.0.14        04:22    0.00s  0.04s  0.00s w` };
  }

  // Commands whose output names the beacon's PID/socket — keep them in sync
  // with the live (possibly respawned) process after a kill.
  const BEACON_VIEW = new Set(['ss', 'lsof', 'ps', 'netstat', 'top']);
  if (store.CMDS[trimmed] !== undefined) {
    return { output: BEACON_VIEW.has(cmd) ? beaconize(store.CMDS[trimmed], state) : store.CMDS[trimmed] };
  }
  if (store.CMDS[cmd] !== undefined) {
    return { output: BEACON_VIEW.has(cmd) ? beaconize(store.CMDS[cmd], state) : store.CMDS[cmd] };
  }

  if (cmd === 'ps' && rest.length === 0) return { output: beaconize(store.CMDS['ps aux'] || '', state) };
  if (cmd === 'netstat' && rest.length === 0) return { output: beaconize(store.CMDS['netstat -an'] || 'Active Internet connections (servers and established)', state) };
  if (cmd === 'ss' && rest.length === 0) return { output: beaconize(store.CMDS['ss -tunap'] || store.CMDS['ss -tnp'] || '', state) };
  if (cmd === 'df' && rest.length === 0) return { output: store.CMDS['df -h'] || '' };
  if (cmd === 'lsof' && rest.length === 0) return { output: beaconize(store.CMDS['lsof -i'] || 'COMMAND PID USER FD TYPE DEVICE SIZE/OFF NODE NAME', state) };
  if (cmd === 'top') return { output: beaconize((store.CMDS['ps aux'] || '').split('\n').slice(0, 12).join('\n'), state) };
  if (cmd === 'nmap') return { output: rest[0] ? `Starting Nmap 7.80\nNmap scan report for ${rest[0]}\nHost is up.\nPORT   STATE SERVICE\n22/tcp open  ssh` : 'Usage: nmap <target>' };
  if (cmd === 'ping') return { output: rest[0] ? (store.CMDS[`ping ${rest[0]}`] || `PING ${rest[0]} (${rest[0]}) 56(84) bytes of data.\n64 bytes from ${rest[0]}: icmp_seq=1 ttl=64 time=2.1 ms\n\n--- ${rest[0]} ping statistics ---\n1 packets transmitted, 1 received, 0% packet loss`) : 'usage: ping host' };

  const deny = (store.SERVER === 'node083' ? DENY_NODE083 : DENY_CORE01)[cmd];
  if (deny) {
    const target = rest.find(a => !a.startsWith('-')) || '?';
    const m = deny(target, rest);
    if (m !== null) return { output: m };
  }

  return { output: `bash: ${cmd}: command not found` };
}

function lsOutput(store, path, flags) {
  if (store.SERVER === 'node083' && path === '/root/databases') {
    if (/-[a-z]*[lh]/.test(flags)) return store.DB_LISTING;
    const names = store.FS['/root/databases'].ch;
    return /-[a-z]*a/.test(flags) ? ['.', '..', ...names].join('  ') : names.join('  ');
  }
  const showHidden = /-[a-z]*a/.test(flags);
  const showDetail = /-[a-z]*l/.test(flags);
  const entry = getEntry(store.FS, path);
  const content = fileContent(store, path);
  if (!entry && content !== undefined) {
    const name = path.split('/').pop();
    if (!showDetail) return name;
    const meta = (store.LS_META || {})[name] || '-rw-r--r--  1 root root     4096 May 17 04:03';
    return `${meta} ${name}`;
  }
  if (!entry) return null;
  if (entry.type === 'file') {
    const name = path.split('/').pop();
    if (!showDetail) return name;
    const meta = (store.LS_META || {})[name] || '-rw-r--r--  1 root root      512 May 17 03:00';
    return `${meta} ${name}`;
  }
  let children = [...(entry.ch || [])];
  if (!showHidden) children = children.filter(c => !c.startsWith('.'));
  children.sort();
  if (!showDetail) return (showHidden ? ['.', '..', ...children] : children).join('  ') || '';
  const lines = [];
  if (showHidden) {
    lines.push('drwxr-xr-x 18 root root  4096 May 17 04:22 .');
    lines.push('drwxr-xr-x 18 root root  4096 May 17 03:00 ..');
  }
  for (const c of children) {
    const meta = (store.LS_META || {})[c] || '-rw-r--r--  1 root root      512 May 17 03:00';
    lines.push(`${meta} ${c}`);
  }
  return lines.join('\n');
}

function runSqlite({ store, raw, state }) {
  const t = raw.trim();
  if (t === '.exit' || t === '.quit' || t === '.q') {
    state.sqliteMode = false; state.sqliteFile = null;
    return { output: '', state };
  }
  if (t === '.tables') return { output: 'entries             offshore_transfers', state };
  if (t === '.schema' || t === '.schema entries' || t === '.schema offshore_transfers') {
    return { output:
`CREATE TABLE "offshore_transfers" (
  "id" INTEGER, "quarter" TEXT, "destination_node" TEXT,
  "amount_usd" TEXT, "client_alias" TEXT, "notes" TEXT
);
CREATE TABLE "entries" (
  "id" INTEGER PRIMARY KEY AUTOINCREMENT,
  "timestamp" TEXT NOT NULL,
  "classification" TEXT DEFAULT 'CLASSIFIED',
  "content" TEXT
);`, state };
  }
  if (t === '.help') return { output:
`.tables         List names of tables
.schema [NAME]  Show CREATE statements
.exit / .quit   Exit SQLite
SELECT * FROM entries;             Read the calling-card record
SELECT * FROM offshore_transfers;  Read the dirty ledger (sample)`, state };
  if (/select/i.test(t) && /offshore_transfers/i.test(t))
    return { output: store.SELECT_OFFSHORE, state };
  if (/select/i.test(t) && /entries/i.test(t))
    return { output: store.SELECT_ENTRIES, state };
  if (/select/i.test(t)) return { output: 'Error: no such table or column referenced', state };
  return { output: `Error: near "${t.split(' ')[0]}": syntax error`, state };
}
